In the code, there was only five led blinking so I made them all blink by added the 3 others.


I have defined 3 more led in this code, from : 

#define LED_PAD					0
#define LED1_PAD				2
#define LED2_PAD				3
#define LED3_PAD				4
#define LED4_PAD				5

to : 

#define LED_PAD					0
#define LED1_PAD				2
#define LED2_PAD				3
#define LED3_PAD				4
#define LED4_PAD				5
#define LED5_PAD				1
#define LED6_PAD				6
#define LED7_PAD				7

After this, I then added 3 more BlinkAgent with the task priority, from : 

void mainTask(void *params){
	BlinkAgent blink(LED_PAD);
	BlinkAgent worker1(LED1_PAD);
	BlinkAgent worker2(LED2_PAD);
	BlinkAgent worker3(LED3_PAD);
	BlinkAgent worker4(LED4_PAD);

	printf("Main task started\n");

	blink.start("Blink", TASK_PRIORITY);
	worker1.start("Worker 1", TASK_PRIORITY);
	worker2.start("Worker 2", TASK_PRIORITY + 1);
	worker3.start("Worker 3", TASK_PRIORITY + 2);
	worker4.start("Worker 4", TASK_PRIORITY + 3);
